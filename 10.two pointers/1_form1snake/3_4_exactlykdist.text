❌ Using:
if(dcnt == k)
    ans += (head - tail + 1);
This counts too many subarrays. It adds every time the current window has exactly k distinct elements — including overlapping subarrays like [1,2,1], [1,2,1,2], etc., even if they don’t start at tail.

👉 So this approach overcounts.
-------------------------------------------------------------------------------
✅ Correct approach:

    Longest Substring (At Most k = matlab (<k) bhi chalega but longest hona chaiye ) --->  ans = max(ans, head - tail + 1);

    Total No. of subarray (At Most k) ---> ans += (head - tail + 1);

    Total No. of subarray (exactly k) ---> ans += (head - tail + 1);---> return solve(k)-solve(k-1);

--------------------------------------------------------------------------------
exactlyK = atMostK(k) - atMostK(k - 1)
atMostK(k) = all subarrays with ≤ k distinct

atMostK(k-1) = all subarrays with ≤ k-1 distinct

Their difference = subarrays with exactly k distinct

This automatically handles all valid subarrays, without double-counting or missing any.

----------------------------------------------------------------------------



#include <bits/stdc++.h>
using namespace std;

int n,k;
//solution ds
int freq[1000100]{};
int dcnt=0;//distinct count 

int main() {
    ios_base::sync_with_stdio(0);
    cin.tie(0);
    cout.tie(0);
    cin>>n>>k;
    int arr[n];
    for(int i=0;i<n;i++){
        cin>>arr[i];
    }
    int head=-1,tail=0;

    int ans=0;
    while(tail<n){
        while(head+1<n && ((freq[arr[head+1]]==0 && dcnt<k) ||
                                 (freq[arr[head+1]]>0 && dcnt<=k))){
            head++;
            //maintain ds
            if(freq[arr[head]]==0)dcnt++;
            freq[arr[head]]++;
        }
        //process ans and ds
        if(dcnt==k){
            ans += (head-tail+1);//just removed max
        }
        //tail++
        if(tail<=head){
            freq[arr[tail]]--;
            if(freq[arr[tail]]==0)dcnt--;
            tail++;
        }else{
            tail++;
            head=tail-1;
        }
    }
    cout<<ans<<"\n";
    return 0;
}
